def create_dataset(dataset, time_step=1):
    """
    Tạo tập dữ liệu đầu vào (X) và đầu ra (Y) theo cửa sổ trượt (sliding window).
    dataset: Mảng 1 chiều chứa chuỗi thời gian.
    time_step: Số bước thời gian dùng để dự báo bước tiếp theo.
    """
    dataX, dataY = [], []
    for i in range(len(dataset) - time_step):
        a = dataset[i:(i + time_step)]
        dataX.append(a)
        dataY.append(dataset[i + time_step])
    return np.array(dataX), np.array(dataY)

def predict_future(model, last_sequence, n_days, scaler):
    """
    Dự đoán n_days tiếp theo từ chuỗi đầu vào cuối cùng (đã chuẩn hóa),
    sau đó chuyển kết quả về giá trị gốc bằng scaler.
    """
    future_preds_scaled = []
    # Đảm bảo current_seq có dạng (1, time_step, 1)
    current_seq = np.array(last_sequence).reshape(1, -1, 1)

    for _ in range(n_days):
        next_pred = model.predict(current_seq, verbose=0)[0, 0]
        future_preds_scaled.append(next_pred)
        # Trượt cửa sổ: bỏ phần tử đầu tiên, thêm phần tử vừa dự báo vào cuối
        next_val = np.array([[[next_pred]]])
        current_seq = np.concatenate([current_seq[:, 1:, :], next_val], axis=1)

    future_preds_scaled = np.array(future_preds_scaled).reshape(-1, 1)
    future_preds_actual = scaler.inverse_transform(future_preds_scaled)
    return future_preds_actual.flatten()

def create_dataset_multivariate(dataset, target_idx=2, time_step=10):
    """
    Tạo cặp (X, Y) cho bài toán chuỗi thời gian đa biến.
    dataset: Mảng 2D có kích thước (samples, num_features).
    target_idx: Chỉ số cột mục tiêu cần dự báo (cột Close có index = 2).
    """
    dataX, dataY = [], []
    for i in range(len(dataset) - time_step):
        dataX.append(dataset[i:(i + time_step), :])
        dataY.append(dataset[i + time_step, target_idx])
    return np.array(dataX), np.array(dataY)
